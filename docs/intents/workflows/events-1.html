        "title": "{{bind.ev.title}}",

{
  "version": 2,
  "bindings": {
    "ref": {
      "type": "eventref",
      "from": "{{$1|default(\"\")}}",
      "reject": [
        "все",

        "обе"
      ],
      "stem": true
    }
  },
  "steps": [
    {
      "when": "bind.ref.kind == 'id'",
      "call": "get_event",
      "input": {
        "event_id": "{{bind.ref.id}}",
        "scope": "{{env.scope}}"
      },
      "as": "target"
    },
    {
      "when": "bind.ref.kind == 'name'",
      "call": "search_events",
      "input": {
        "query": "{{bind.ref.query}}",
        "scope": "{{env.scope}}",
        "event_type": "regular"
      },
      "as": "found"
    },
    {
      "when": "count(found) == 0 && bind.ref.kind == 'name'",
      "abstain": true
    },
    {
      "when": "count(found) > 1",
      "respond": "{{t.ambiguous}}"
    },
    {
      "when": "count(found) == 1",

  ],
  "i18n": {
