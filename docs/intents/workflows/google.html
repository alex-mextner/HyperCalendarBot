Как подключить, переподключить или отключить Google Календарь

{
  "version": 2,
  "steps": [
    {
      "respond": "{{t.help}}"
    }
  ],
  "i18n": {
    "ru": {
      "help": "Google Calendar подключается командой /connect_google, отключается командой /disconnect_google — обе работают в личке с ботом. По фразе в чате я ничего не отключаю."
    },
    "en": {
      "help": "Google Calendar is connected with /connect_google and disconnected with /disconnect_google — both work in a private chat with the bot. I do not disconnect anything from a chat phrase."

Статус подключённого личного Telegram

{
  "version": 2,
  "steps": [
    {
      "when": "group.is_group == true",
      "respond": "{{t.private_only}}"
    },
    {
      "call": "connect_telegram_status",
      "input": {},
      "as": "connection"
    },
    {
      "respond": "{{tool_outputs.connection_text}}\n\n{{t.connect_help}}"
    }
  ],
  "i18n": {
    "ru": {
      "private_only": "Это личные данные — работаю с ними только в личке с ботом.",
      "connect_help": "Для подключения вашего аккаунта откройте /connect_telegram в личном чате с ботом. Сам статус подключения ничего не меняет."
    },
    "en": {
      "private_only": "This is private data — I only handle it in a private chat with the bot.",
      "connect_help": "To connect your account, open /connect_telegram in this private bot chat. Reading the status does not change the connection."
    }
  }
}
