Когда или во сколько событие

basis.calendar.when · Back to rule

{
  "version": 2,
  "bindings": {
    "ref": {
      "type": "eventref",
      "from": "{{$1}}",
      "stem": true
    }
  },
  "steps": [
    {
      "when": "bind.ref.kind == 'id'",
      "abstain": true
    },
    {
      "call": "search_events",
      "input": {
        "query": "{{bind.ref.query}}",
        "scope": "{{env.scope}}"
      },
      "as": "found"
    },
    {
      "when": "count(found) == 0",
      "abstain": true
    },
    {
      "when": "count(found) == 1 && found[0].all_day == false",
      "respond": "{{t.one}}"
    },
    {
      "when": "count(found) == 1 && found[0].all_day == true",
      "respond": "{{t.one_all_day}}"
    }
  ],
  "i18n": {
    "ru": {
      "one": "«{{tool_outputs.found[0].title}}» — {{tool_outputs.found[0].date|human_day}} в {{tool_outputs.found[0].time}}.",

    },
    "en": {
      "one": "“{{tool_outputs.found[0].title}}” — {{tool_outputs.found[0].date|human_day}} at {{tool_outputs.found[0].time}}.",
